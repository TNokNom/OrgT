-> A Neural Network essentially takes input data and multiplies it by a weight
and outputs a prediction
-> Weights can be looked at as sensitivity knob
#+Begin_Src python :results output
weight = 0.1
def neural_netowrk(input, weight):
    prediction = input * weight
    return prediction
number_of_toes = [ 8.5, 9.5, 10, 9]
input = number_of_toes[0]
pred = neural_netowrk(input,weight)
print (pred)
