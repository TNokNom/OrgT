(defun my-org-parentheses-color ()
  "Customize the color of all parentheses in Org mode."
  (font-lock-add-keywords
   nil
   '(("[(|)]" . 'my-org-paren-face))))

(defface my-org-paren-face
  '((t (:foreground "#ea6962")))  ;; Change "orange" to any color you prefer
  "Face for parentheses in Org mode.")

(add-hook 'org-mode-hook #'my-org-parentheses-color)

(defun my-org-arrow-color ()
  "Customize the color of all parentheses in Org mode."
  (font-lock-add-keywords
   nil
   '(("[->]" . 'my-org-paren-face))))

  '((t (:foreground "#ea6962")))  ;; Change "orange" to any color you prefer
